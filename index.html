/**
 * QR Code Attendance — Google Sheets relay
 * ==========================================
 * Deploy this INSIDE the Google Sheet you want to use as your shared
 * attendance file:
 *   1. Open the target Google Sheet.
 *   2. Extensions → Apps Script.
 *   3. Delete anything in the editor, paste this whole file in.
 *   4. Deploy → New deployment → gear icon → "Web app".
 *        Execute as: Me
 *        Who has access: Anyone
 *   5. Deploy → copy the Web app URL → paste it into the app's
 *      Settings → "Google Sheets sync" field.
 *
 * Sheet layout this expects (matches the app's own Excel export exactly):
 *   Row 1: title cells (event name, merged per date-pair, blank over Name/ID)
 *   Row 2: headers — Name, Student ID, then "<date> - Time In" / "<date> - Time Out" pairs
 *   Row 3+: one row per student
 *
 * Each worksheet TAB in the spreadsheet is treated as one "group" in the app.
 * Add a new tab in Sheets (with at least "Name"/"Student ID" in row 2, or
 * just leave it blank — the relay fills that in) and hit "Refresh" in the
 * app to pick it up. Deleting a group in the app does NOT delete the sheet
 * tab — it only stops tracking it on that device.
 */

function doGet(e) {
  var action = e.parameter.action;
  if (action === 'snapshot') {
    return respond(buildSnapshot());
  }
  return respond({ ok: false, error: 'Unknown action: ' + action });
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  var gotLock = lock.tryLock(15000);
  if (!gotLock) {
    return respond({ ok: false, error: 'Sheet is busy, try again in a moment.' });
  }
  try {
    var payload = JSON.parse(e.postData.contents);
    var action = payload.action;
    if (action === 'logAttendance') return respond(logAttendance(payload));
    if (action === 'deleteColumn') return respond(deleteColumnAction(payload));
    if (action === 'addStudent') return respond(addStudentAction(payload));
    if (action === 'clearCell') return respond(clearCellAction(payload));
    return respond({ ok: false, error: 'Unknown action: ' + action });
  } catch (err) {
    return respond({ ok: false, error: String(err) });
  } finally {
    lock.releaseLock();
  }
}

function respond(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function getSS() {
  return SpreadsheetApp.getActiveSpreadsheet();
}

/* ---------- Colors (matches the app's local Excel export exactly) ---------- */
var COLOR_PRIMARY      = '#0B6E4F'; // header row background, title text
var COLOR_PRIMARY_TINT = '#DFF3EA'; // title row background
var COLOR_WHITE         = '#FFFFFF';
var COLOR_PRESENT_BG    = '#C6EFCE'; // Time In cell background
var COLOR_PRESENT_FG    = '#006100'; // Time In text
var COLOR_OUT_BG        = '#FBE7E5'; // Time Out cell background
var COLOR_OUT_FG        = '#C5413B'; // Time Out text

/* ---------- Sheet layout helpers ---------- */

function ensureHeaderRows(sheet) {
  if (sheet.getLastRow() < 2) {
    sheet.getRange(1, 1, 1, 2).setValues([['', '']]).setBackground(COLOR_WHITE);
    sheet.getRange(2, 1, 1, 2).setValues([['Name', 'Student ID']])
      .setBackground(COLOR_PRIMARY)
      .setFontColor(COLOR_WHITE)
      .setFontWeight('bold')
      .setHorizontalAlignment('center')
      .setVerticalAlignment('middle');
    sheet.setColumnWidth(1, 210); // Name
    sheet.setColumnWidth(2, 100); // Student ID
    sheet.setFrozenRows(2);
    sheet.setFrozenColumns(2);
  }
}

function getHeaderRow(sheet) {
  var lastCol = Math.max(sheet.getLastColumn(), 2);
  return sheet.getRange(2, 1, 1, lastCol).getValues()[0];
}

function findDateColIndex(headerRow, sessionLabel, suffix) {
  var target = sessionLabel + ' - ' + suffix;
  for (var i = 0; i < headerRow.length; i++) {
    if (headerRow[i] === target) return i + 1; // 1-indexed column
  }
  return -1;
}

// Returns the 1-indexed column of "<sessionLabel> - Time In" (Time Out is
// always the column right after it), creating both columns + the title
// banner cell above them if this date hasn't been logged yet.
function ensureDateColumns(sheet, sessionLabel, eventName) {
  ensureHeaderRows(sheet);
  var headerRow = getHeaderRow(sheet);
  var inCol = findDateColIndex(headerRow, sessionLabel, 'Time In');
  if (inCol !== -1) return inCol;

  var newCol = sheet.getLastColumn() + 1;
  if (newCol < 3) newCol = 3;

  sheet.getRange(2, newCol, 1, 2).setValues([[sessionLabel + ' - Time In', sessionLabel + ' - Time Out']])
    .setBackground(COLOR_PRIMARY)
    .setFontColor(COLOR_WHITE)
    .setFontWeight('bold')
    .setHorizontalAlignment('center')
    .setVerticalAlignment('middle');
  sheet.setColumnWidth(newCol, 90);
  sheet.setColumnWidth(newCol + 1, 90);

  var titleRange = sheet.getRange(1, newCol, 1, 2);
  titleRange.setValues([[eventName || 'Attendance', '']])
    .setBackground(COLOR_PRIMARY_TINT)
    .setFontColor(COLOR_PRIMARY)
    .setFontWeight('bold')
    .setHorizontalAlignment('center')
    .setVerticalAlignment('middle');
  try { titleRange.merge(); } catch (mergeErr) { /* fine if it can't merge */ }

  return newCol;
}

function findStudentRow(sheet, studentId) {
  var lastRow = sheet.getLastRow();
  if (lastRow < 3) return -1;
  var ids = sheet.getRange(3, 2, lastRow - 2, 1).getValues();
  for (var i = 0; i < ids.length; i++) {
    if (String(ids[i][0]).trim() === String(studentId).trim()) return i + 3;
  }
  return -1;
}

/* ---------- Actions ---------- */

function logAttendance(p) {
  var sheet = getSS().getSheetByName(p.group);
  if (!sheet) return { ok: false, error: 'No tab named "' + p.group + '" in this sheet.' };

  var col = ensureDateColumns(sheet, p.sessionLabel, p.eventName);
  var targetCol = (p.mode === 'out') ? col + 1 : col;

  var row = findStudentRow(sheet, p.studentId);
  if (row === -1) {
    row = Math.max(sheet.getLastRow() + 1, 3);
    sheet.getRange(row, 1).setValue(p.studentName || '');
    sheet.getRange(row, 2).setValue(p.studentId).setFontFamily('Consolas');
  } else if (p.studentName) {
    sheet.getRange(row, 1).setValue(p.studentName);
  }

  var cell = sheet.getRange(row, targetCol);
  var existing = cell.getValue();
  if (existing) {
    return { ok: true, duplicate: true, existing: existing };
  }
  cell.setValue(p.time).setHorizontalAlignment('center');
  if (p.mode === 'out') {
    cell.setBackground(COLOR_OUT_BG).setFontColor(COLOR_OUT_FG);
  } else {
    cell.setBackground(COLOR_PRESENT_BG).setFontColor(COLOR_PRESENT_FG);
  }
  return { ok: true, duplicate: false };
}

function clearCellAction(p) {
  var sheet = getSS().getSheetByName(p.group);
  if (!sheet) return { ok: false, error: 'No tab named "' + p.group + '" in this sheet.' };

  var headerRow = getHeaderRow(sheet);
  var inCol = findDateColIndex(headerRow, p.sessionLabel, 'Time In');
  if (inCol === -1) return { ok: false, error: 'That date column was not found.' };
  var targetCol = (p.mode === 'out') ? inCol + 1 : inCol;

  var row = findStudentRow(sheet, p.studentId);
  if (row === -1) return { ok: false, error: 'That student was not found in this group.' };

  sheet.getRange(row, targetCol).clear(); // clears value + background + font color together
  return { ok: true };
}

function deleteColumnAction(p) {
  var sheet = getSS().getSheetByName(p.group);
  if (!sheet) return { ok: false, error: 'No tab named "' + p.group + '" in this sheet.' };
  var headerRow = getHeaderRow(sheet);
  var col = findDateColIndex(headerRow, p.sessionLabel, 'Time In');
  if (col === -1) return { ok: false, error: 'That date column was not found.' };
  sheet.deleteColumns(col, 2);
  return { ok: true };
}

function addStudentAction(p) {
  var sheet = getSS().getSheetByName(p.group);
  if (!sheet) return { ok: false, error: 'No tab named "' + p.group + '" in this sheet.' };
  ensureHeaderRows(sheet);
  var row = findStudentRow(sheet, p.studentId);
  if (row === -1) {
    row = Math.max(sheet.getLastRow() + 1, 3);
    sheet.getRange(row, 1).setValue(p.studentName || '');
    sheet.getRange(row, 2).setValue(p.studentId).setFontFamily('Consolas');
  } else if (p.studentName) {
    sheet.getRange(row, 1).setValue(p.studentName);
  }
  return { ok: true };
}

function buildSnapshot() {
  var sheets = getSS().getSheets();
  var groups = {};

  sheets.forEach(function (sheet) {
    var name = sheet.getName();
    var lastRow = sheet.getLastRow();
    var lastCol = sheet.getLastColumn();

    if (lastRow < 2 || lastCol < 2) {
      groups[name] = { dateColumns: [], eventNames: {}, roster: [] };
      return;
    }

    var headerRow = sheet.getRange(2, 1, 1, lastCol).getValues()[0];
    var titleRow = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
    var dateColumns = [];
    var eventNames = {};

    // Columns are 0-indexed here; data columns start at index 2 (col C) and
    // come in Time In/Time Out pairs.
    for (var c = 2; c < lastCol; c += 2) {
      var inHeader = headerRow[c];
      if (!inHeader) continue;
      var label = String(inHeader).replace(/ - Time In$/, '');
      dateColumns.push(label);
      eventNames[label] = titleRow[c] || 'Attendance';
    }

    var roster = [];
    if (lastRow >= 3) {
      var data = sheet.getRange(3, 1, lastRow - 2, lastCol).getValues();
      data.forEach(function (row) {
        var studentName = row[0];
        var studentId = row[1];
        if (!studentId) return;
        var attendance = {};
        dateColumns.forEach(function (label, idx) {
          var inVal = row[2 + idx * 2];
          var outVal = row[2 + idx * 2 + 1];
          if (inVal || outVal) {
            attendance[label] = { in: inVal || '', out: outVal || '' };
          }
        });
        roster.push({ id: String(studentId), name: String(studentName || ''), attendance: attendance });
      });
    }

    groups[name] = { dateColumns: dateColumns, eventNames: eventNames, roster: roster };
  });

  return { ok: true, groups: groups };
}
